set(YMGRE_CAMERA_COLOR_DEPTH 24 CACHE STRING "Camera framebuffer color depth (24 RGB888 for PC, 16 RGB565 for MCU)")
set_property(CACHE YMGRE_CAMERA_COLOR_DEPTH PROPERTY STRINGS 16 24)
if(NOT YMGRE_CAMERA_COLOR_DEPTH STREQUAL "16" AND NOT YMGRE_CAMERA_COLOR_DEPTH STREQUAL "24")
	message(FATAL_ERROR "YMGRE_CAMERA_COLOR_DEPTH must be 16 or 24")
endif()

set(YMGRE_INDEX_BITS 16 CACHE STRING "Mesh index width (16 for MCU, 32 for large meshes)")
set_property(CACHE YMGRE_INDEX_BITS PROPERTY STRINGS 16 32)
if(NOT YMGRE_INDEX_BITS STREQUAL "16" AND NOT YMGRE_INDEX_BITS STREQUAL "32")
	message(FATAL_ERROR "YMGRE_INDEX_BITS must be 16 or 32")
endif()

# All runnable source demos share one output layout. Tests stay in their build tree.
set(YMGRE_PROGRAM_OUTPUT_DIR "" CACHE PATH "Override the executable directory for source demos")
if(YMGRE_PROGRAM_OUTPUT_DIR)
 set(_ymgre_program_dir "${YMGRE_PROGRAM_OUTPUT_DIR}")
else()
 if(YMGRE_CAMERA_COLOR_DEPTH STREQUAL "16")
  set(_ymgre_program_format rgb565)
 else()
  set(_ymgre_program_format rgb888)
 endif()
 set(_ymgre_program_dir "${YMGRE_REPO_ROOT}/build/bin/${_ymgre_program_format}/index${YMGRE_INDEX_BITS}")
endif()
set_property(GLOBAL PROPERTY YMGRE_PROGRAM_DIRECTORY "${_ymgre_program_dir}")
function(ymgre_set_program_output)
 get_property(_output GLOBAL PROPERTY YMGRE_PROGRAM_DIRECTORY)
 foreach(_target IN LISTS ARGN)
  set_target_properties(${_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${_output}")
 endforeach()
endfunction()
