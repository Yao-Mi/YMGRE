cmake_minimum_required(VERSION 3.16)
project(YMGRE_GirlViewer C)
set(CMAKE_C_STANDARD 99)
set(YMGRE_INDEX_BITS 32)
set(YMGRE_CAMERA_COLOR_DEPTH 24)
foreach(feature PBR TRANSPARENCY OPACITY_MIPMAP LINEAR_COLOR RASTER_DISPATCH PNG JPEG)
 set(YMGRE_ENABLE_${feature} ON)
endforeach()
if(NOT YMGRE_SDK_ROOT)
 message(FATAL_ERROR "Pass -DYMGRE_SDK_ROOT=/path/to/YMGRE_libs (full profile)")
endif()
find_package(YMGRE CONFIG REQUIRED PATHS "${YMGRE_SDK_ROOT}/cmake" NO_DEFAULT_PATH)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED sdl2 SDL2_ttf)
add_executable(girl_viewer main.c raster_pool.c)
set(YMGRE_GIRL_ASSET_DIR "" CACHE PATH "Optional directory containing the girl mesh/material/texture files")
if(YMGRE_GIRL_ASSET_DIR)
 target_compile_definitions(girl_viewer PRIVATE YMGRE_GIRL_ASSET_DIR="${YMGRE_GIRL_ASSET_DIR}")
endif()
target_include_directories(girl_viewer PRIVATE ${SDL_INCLUDE_DIRS})
target_link_libraries(girl_viewer PRIVATE YMGRE::ymgre ${SDL_LIBRARIES})

set(YMGRE_PROGRAM_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/bin/rgb888/index32" CACHE PATH "Girl executable output directory")
set_target_properties(girl_viewer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${YMGRE_PROGRAM_OUTPUT_DIR}")
