cmake_minimum_required(VERSION 3.16)
project(lod_mesh_preview C)
set(CMAKE_C_STANDARD 99)
set(YMGRE_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(YMGRE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YMGRE_BUILD_YMGUI_HOST OFF CACHE BOOL "" FORCE)
get_filename_component(YMGRE_REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${YMGRE_REPO}" "${CMAKE_BINARY_DIR}/ymgre")
add_executable(lod_mesh_preview lod_mesh_preview.c)
target_link_libraries(lod_mesh_preview PRIVATE ymgre)
target_compile_definitions(lod_mesh_preview PRIVATE RESOURCE_ROOT="${YMGRE_REPO}/Resource")
target_compile_options(lod_mesh_preview PRIVATE -Wall -Wextra)
enable_testing()
add_test(NAME lod_mesh_preview_smoke COMMAND lod_mesh_preview "${CMAKE_BINARY_DIR}/lod_mesh_preview.bmp")
add_test(NAME lod_mesh_scene_smoke COMMAND lod_mesh_preview --scene "${CMAKE_BINARY_DIR}/lod_mesh_scene.bmp")
add_test(NAME lod_distance_grass_smoke COMMAND lod_mesh_preview --distance-grass "${CMAKE_BINARY_DIR}/lod_distance_grass.bmp")
add_test(NAME lod_distance_tree_smoke COMMAND lod_mesh_preview --distance-tree "${CMAKE_BINARY_DIR}/lod_distance_tree.bmp")
add_test(NAME lod_basic_runtime_smoke COMMAND lod_mesh_preview --basic "${CMAKE_BINARY_DIR}/lod_basic")
add_test(NAME lod_basic_no_mip_smoke COMMAND lod_mesh_preview --basic-no-mip "${CMAKE_BINARY_DIR}/lod_basic_no_mip")
