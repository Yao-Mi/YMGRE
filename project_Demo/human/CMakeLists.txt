cmake_minimum_required(VERSION 3.16)
project(YMGRE_Human_Demo C)
set(CMAKE_C_STANDARD 99)
get_filename_component(GRE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(YMSKE_ROOT "${GRE_ROOT}/../YMSKE" CACHE PATH "YMSKE source directory")
set(YMGRE_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(YMGRE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YMGRE_BUILD_YMGUI_HOST ON CACHE BOOL "" FORCE)
add_subdirectory("${GRE_ROOT}" gre)
set(YMSKE_BUILD_SKIN ON CACHE BOOL "" FORCE)
set(YMSKE_BUILD_IO ON CACHE BOOL "" FORCE)
set(YMSKE_BUILD_GRE ON CACHE BOOL "" FORCE)
set(YMSKE_BUILD_PME OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_HAND_DEMO OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_OWL_DEMO OFF CACHE BOOL "" FORCE)
set(YMSKE_MAX_INFLUENCES 6 CACHE STRING "" FORCE)
set(YMSKE_GRE_ROOT "${GRE_ROOT}" CACHE PATH "" FORCE)
add_subdirectory("${YMSKE_ROOT}" ske)
add_library(human_gre_file_support OBJECT "${GRE_ROOT}/YMGRE/CORE/YMGRE_SceneManager.c")
target_link_libraries(human_gre_file_support PRIVATE ymgre)
add_executable(demo_human main.c scene.c $<TARGET_OBJECTS:human_gre_file_support>)
target_compile_options(demo_human PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(demo_human PRIVATE HUMAN_ASSET_ROOT="${GRE_ROOT}/Resource/human")
get_target_property(human_gui_includes ymgui INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(demo_human SYSTEM PRIVATE ${human_gui_includes})
target_link_libraries(demo_human PRIVATE YMSKE::gre YMSKE::io ymgre_demo_host)
enable_testing()
add_test(NAME human_headless COMMAND demo_human --headless)
add_test(NAME human_window COMMAND demo_human --smoke-test)
set_tests_properties(human_window PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 30)
