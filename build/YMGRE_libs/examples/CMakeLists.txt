cmake_minimum_required(VERSION 3.16)
project(YMGRE_Binary_Examples C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(YMGRE_INDEX_BITS 16 CACHE STRING "Precompiled index width")
option(YMGRE_WINDOWED_DEMOS "Build window examples using an external YMGUI SDK" OFF)
if(NOT YMGRE_DIR)
    set(YMGRE_DIR "${CMAKE_CURRENT_LIST_DIR}/../cmake")
endif()
find_package(YMGRE CONFIG REQUIRED)
if(YMGRE_WINDOWED_DEMOS)
    # The YMGUI SDK owns its headers, libraries and platform dependencies.
    find_package(YMGUI CONFIG REQUIRED)
    set(YMGRE_YMGUI_TARGET "YMGUI::ymgui" CACHE STRING "External YMGUI core target")
    set(YMGRE_YMGUI_SDL_TARGET "YMGUI::sdl" CACHE STRING "External YMGUI SDL target")
    foreach(_dependency IN ITEMS "${YMGRE_YMGUI_TARGET}" "${YMGRE_YMGUI_SDL_TARGET}")
        if(NOT TARGET "${_dependency}")
            message(FATAL_ERROR "External YMGUI SDK must export ${_dependency}; override YMGRE_YMGUI_TARGET / YMGRE_YMGUI_SDL_TARGET to match its target names")
        endif()
    endforeach()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(YMGRE_SDL2 REQUIRED IMPORTED_TARGET sdl2)
    # Our small example adapter is application source, not an engine dependency.
    add_library(ymgre_example_host STATIC host/demo_host.c)
    target_include_directories(ymgre_example_host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/host")
    target_link_libraries(ymgre_example_host PUBLIC YMGRE::ymgre
        "${YMGRE_YMGUI_TARGET}" "${YMGRE_YMGUI_SDL_TARGET}" PkgConfig::YMGRE_SDL2)
endif()
enable_testing()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
file(GLOB _examples CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/demo_*.c")
foreach(_source IN LISTS _examples)
    get_filename_component(_name "${_source}" NAME_WE)
    if(NOT YMGRE_WINDOWED_DEMOS AND NOT _name STREQUAL "demo_sdk_minimal")
        continue()
    endif()
    add_executable(${_name} "${_source}")
    if(_name STREQUAL "demo_sdk_minimal")
        target_link_libraries(${_name} PRIVATE YMGRE::ymgre)
    else()
        target_link_libraries(${_name} PRIVATE ymgre_example_host)
    endif()
    add_test(NAME ${_name} COMMAND ${_name})
    set_tests_properties(${_name} PROPERTIES TIMEOUT 30
        ENVIRONMENT "YMGRE_HEADLESS=1;YMGRE_MAX_FRAMES=1;YMGRE_WINDOW_SCALE=1")
endforeach()
