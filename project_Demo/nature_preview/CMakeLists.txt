cmake_minimum_required(VERSION 3.16)
project(nature_preview C)
set(YMGRE_INDEX_BITS 32 CACHE STRING "Desktop preview mesh index width")
include(${CMAKE_CURRENT_LIST_DIR}/../ymgre_app.cmake)
add_executable(nature_preview nature_preview.c grass_impostor.c)
target_link_libraries(nature_preview PRIVATE ymgre ymgre_demo_host)
target_compile_definitions(nature_preview PRIVATE NATURE_ROOT="${CMAKE_CURRENT_LIST_DIR}" RESOURCE_ROOT="${YMGRE_APP_REPO_ROOT}/Resource")
target_compile_options(nature_preview PRIVATE -Wall -Wextra)
enable_testing()
add_test(NAME nature_preview_camera_smoke COMMAND nature_preview --smoke-test)
set_tests_properties(nature_preview_camera_smoke PROPERTIES ENVIRONMENT "YMGRE_HEADLESS=1" TIMEOUT 60)

add_executable(test_grass_impostor test_grass_impostor.c grass_impostor.c)
target_link_libraries(test_grass_impostor PRIVATE ymgre)
add_test(NAME nature_grass_impostor_depth COMMAND test_grass_impostor)
